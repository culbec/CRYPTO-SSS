\begin{frame}{Conclusion \cite{beimel2011secret}}
	\textbf{\textcolor{blue}{Takeaways}}
	\begin{itemize}
		\item Secret sharing formalizes \textbf{who can reconstruct} via an access structure \(\mathcal{A}\).
		\item Threshold schemes are the baseline for \(t\)-out-of-\(n\) policies.
		\item General policies are possible, but share size and efficiency are the main challenges.
		\item Real deployments may need extensions: robustness (cheating), proactive refresh, etc.
	\end{itemize}
\end{frame}
