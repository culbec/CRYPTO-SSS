\begin{frame}{Enabling Technologies (Math)}
	\begin{itemize}
		\item \textbf{Finite fields} \(\mathbb{F}_q\): arithmetic for Shamir/LSSS (addition, multiplication, inverses).
		\item \textbf{Polynomial interpolation}: efficient reconstruction (Lagrange coefficients).
		\item \textbf{Linear algebra}: matrix-based secret sharing and reasoning about reconstruction.
	\end{itemize}
\end{frame}

\begin{frame}{Enabling Technologies (Systems)}
	\begin{itemize}
		\item \textbf{Secure channels} for distributing shares (TLS/PKI).
		\item \textbf{Reliable randomness} for sampling coefficients/nonces.
		\item \textbf{Distributed storage/replication} across nodes/regions (threat model: partial compromise).
		\item \textbf{Secure hardware} (HSM/TEE) can protect shares-at-rest and limit extraction.
	\end{itemize}
\end{frame}
